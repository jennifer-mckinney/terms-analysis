Data Flow (UML Sequence)

@startuml
actor User
participant "Web App\n(Next.js)" as Web
participant "API Gateway" as APIGW
participant "Ingestion" as Ingest
participant "OCR" as OCR
participant "Normalizer" as Norm
participant "Segmenter" as Seg
participant "Rights Mapper" as Rights
participant "Dark-Pattern Analyzer" as Dark
participant "LLM Reasoner (RAG)" as LLM
database "Embeddings Index" as Vec
database "Legal KB" as KB
folder "Evidence Store" as Obj
database "Results DB" as DB
queue "Event Bus" as Bus
participant "Exporter" as Export
participant "Dashboard Service" as Dash

== Submit Document ==
User -> Web : Paste URL / Upload PDF
Web -> APIGW : POST /analysis
APIGW -> Ingest : startAnalysis(docRef, opts)
Ingest -> Ingest : Snapshot + hash
Ingest -> Obj : PUT policy snapshot (hash)
Ingest -> OCR : OCR if image-based
OCR --> Ingest : text
Ingest -> Norm : normalize(text)
Norm --> Ingest : clean_text
Ingest -> Seg : segment(clean_text)
Seg --> Ingest : sections (rights, ADM, data sale, etc.)

== Analysis ==
par parallel checks
  Ingest -> Rights : map_to_rights(sections, jurisdiction)
  Ingest -> Dark : scan_dark_patterns(sections)
end
Rights --> Ingest : rights_findings
Dark --> Ingest : pattern_findings

Ingest -> LLM : synthesize(findings, excerpts)
LLM -> Vec : query(embeddings)
LLM -> KB : retrieve(citations)
Vec --> LLM : matches
KB --> LLM : citations
LLM --> Ingest : final_findings + citations + scores

Ingest -> DB : save(findings, scores, audit)
DB --> Ingest : id
Ingest -> Bus : publish analysis.completed(id)
Bus -> Dash : update aggregates

== Results & Export ==
APIGW <-- Ingest : analysisId
Web <-- APIGW : 202 Accepted + polling URL
Web -> APIGW : GET /analysis/{id}
APIGW -> DB : fetch(id)
DB --> APIGW : findings+scores
APIGW --> Web : findings panel (with evidence links)

User -> Web : Export (PDF/Word/JSON)
Web -> APIGW : POST /export/{id}
APIGW -> Export : generate(id)
Export -> Obj : write report files
Export --> APIGW : presigned URLs
APIGW --> Web : download links
@enduml

|User|
start
:Open Reviewer UI;
:Paste URL / Upload PDF;
:Select jurisdiction (EU/CA/US baseline);
:Optional: opt-in to contribute anonymized results;

|App UI|
:Validate inputs; show consent text;
:POST analysis request;

|Backend|
:Snapshot policy + hash;
:Normalize & segment content;
:Run rights mapping & dark-pattern checks;
:LLM+RAG synthesis with citations;
:Persist findings & publish event;

|App UI|
:Poll for results;
:Render findings panel (severity, rights, evidence);
:Show uncertainty notes & legal citations;

|User|
:Drill down to excerpts;
:Generate rights request (access/delete/correct/opt-out);
:Export report (PDF/Word/JSON);
if (Share anonymized to rolling dashboard?) then (Yes)
  |Backend|
  :Aggregate into warehouse;
else (No)
endif

|User|
:Provide feedback on accuracy/clarity;
stop
@enduml

Pool("AI Terms Reviewer") {
  Lane("User") {
    StartEvent(start, "Start")
    UserTask(submit, "Submit URL/PDF & select jurisdiction")
  }
  Lane("App UI") {
    Task(validate, "Validate input & consent")
    Task(request, "POST analysis request")
    Task(render, "Render findings & exports")
  }
  Lane("Backend") {
    Task(snapshot, "Snapshot & hash policy")
    Task(normalize, "Normalize & segment content")
    ParallelGateway(split, "Parallel checks")
    Task(maprights, "Map rights (GDPR/AI Act/CPRA)")
    Task(darkscan, "Dark-pattern scan (FTC taxonomy)")
    ParallelGateway(join, "Join")
    Task(synthesize, "LLM+RAG synthesis with citations")
    DataObject(evidence, "Evidence snapshot")
    DataStore(results, "Results DB (findings, audit)")
    IntermediateThrowEvent(pub, "analysis.completed")
  }
  Lane("Analytics") {
    DataStore(wh, "Analytics Warehouse")
    Task(aggregate, "Update rolling aggregates")
    EndEvent(reportend, "Report Ready")
  }
}

SequenceFlow(start, submit)
SequenceFlow(submit, validate)
SequenceFlow(validate, request)
SequenceFlow(request, snapshot)
SequenceFlow(snapshot, normalize)
SequenceFlow(normalize, split)
SequenceFlow(split, maprights)
SequenceFlow(split, darkscan)
SequenceFlow(maprights, join)
SequenceFlow(darkscan, join)
SequenceFlow(join, synthesize)
SequenceFlow(synthesize, results)
SequenceFlow(results, pub)
SequenceFlow(pub, render)
ExclusiveGateway(optin, "Opt-in share?")
SequenceFlow(render, optin)
SequenceFlow(optin, aggregate, "Yes")
SequenceFlow(optin, reportend, "No")
SequenceFlow(aggregate, wh)
SequenceFlow(wh, reportend)
@enduml
